def create_chunks(segments, max_words=150):
    chunks = []
    
    current_text = []
    current_start = None
    current_end = None
    word_count = 0

    for segment in segments:
        text = segment["text"]
        words = len(text.split())

        if current_start is None:
            current_start = segment["start"]

        current_text.append(text)
        current_end = segment["end"]
        word_count += words

        if word_count >= max_words:
            chunks.append({
                "text": " ".join(current_text),
                "start": current_start,
                "end": current_end
            })

            current_text = []
            current_start = None
            current_end = None
            word_count = 0

    # Add remaining text
    if current_text:
        chunks.append({
            "text": " ".join(current_text),
            "start": current_start,
            "end": current_end
        })

    return chunks


chunks = create_chunks(segments, max_words=150)

print("Total chunks:", len(chunks))
print("\nFirst chunk:")
print(chunks[0])

### CELL 12 ###
def format_timestamp(seconds):
    seconds = int(seconds)

    hours = seconds // 3600
    minutes = (seconds % 3600) // 60
    seconds = seconds % 60

    if hours > 0:
        return f"{hours:02d}:{minutes:02d}:{seconds:02d}"
    else:
        return f"{minutes:02d}:{seconds:02d}"

def retrieve_chunks(question, n_results=5):
    
    # Convert user's question into an embedding
    query_embedding = embedding_model.encode(
        [question]
    )

    # Search ChromaDB
    results = collection.query(
        query_embeddings=query_embedding.tolist(),
        n_results=n_results
    )

    return results

def ask_video(question, n_results=5):

    # 1. Retrieve relevant chunks
    results = retrieve_chunks(
        question,
        n_results=n_results
    )

    # 2. Build context for Llama
    context_parts = []

    for i, document in enumerate(results["documents"][0]):

        metadata = results["metadatas"][0][i]

        start_time = format_timestamp(metadata["start"])
        end_time = format_timestamp(metadata["end"])

        context_parts.append(
            f"""
SOURCE {i + 1}
TIMESTAMP: {start_time} - {end_time}

{document}
"""
        )

    context = "\n".join(context_parts)

    # 3. Ask Llama using only retrieved context
    prompt = f"""
You are DeepDive, an AI assistant that answers questions
about a video.

Answer the user's question using ONLY the transcript
context provided below.

If the answer cannot be found in the context,
say: "I couldn't find this information in the video."

Keep the answer concise and accurate.

USER QUESTION:
{question}

VIDEO CONTEXT:
{context}
"""

    # 4. Generate answer using local Llama
    response = ollama.chat(
        model="llama3.1:8b",
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ],
        options={
            "num_gpu": 0,
            "num_predict": 300
        }
    )

    return response["message"]["content"], results
